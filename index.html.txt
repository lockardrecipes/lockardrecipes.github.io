<script>

let recipes = [];
let selectedTags = new Set();

// Find spreadsheet columns using flexible header names.
function getField(row, names) {
    const keys = Object.keys(row);

    for (const name of names) {
        const key = keys.find(k =>
            k.toLowerCase().trim() === name.toLowerCase()
        );

        if (key !== undefined && row[key] != null) {
            return String(row[key]).trim();
        }
    }

    return "";
}

// Convert spreadsheet rows into recipe objects.
function normalizeRecipe(row) {
    const name = getField(row, [
        "Recipe", "Recipe Name", "Name", "Title"
    ]);

    const tagText = getField(row, [
        "Tags", "Tag", "Categories", "Category"
    ]);

    return {
        name: name,
        ingredients: getField(row, [
            "Ingredients", "Ingredient"
        ]),
        instructions: getField(row, [
            "Instructions", "Directions", "Method", "Steps"
        ]),
        description: getField(row, [
            "Description", "Notes", "Summary"
        ]),
        tags: tagText.split(/[,;|]/)
            .map(t => t.trim())
            .filter(Boolean),
        prepTime: getField(row, [
            "Prep Time", "Cooking Time", "Time"
        ]),
        servings: getField(row, [
            "Servings", "Yield"
        ])
    };
}

// Upload and read Excel file.
document.getElementById("excelFile")
.addEventListener("change", function(event) {

    const file = event.target.files[0];

    if (!file) return;

    const reader = new FileReader();

    reader.onload = function(e) {

        try {
            const workbook = XLSX.read(e.target.result, {
                type: "array"
            });

            const sheet = workbook.Sheets[
                workbook.SheetNames[0]
            ];

            const data = XLSX.utils.sheet_to_json(sheet, {
                defval: ""
            });

            recipes = data
                .map(normalizeRecipe)
                .filter(r => r.name);

            selectedTags.clear();

            if (!recipes.length) {
                throw new Error(
                    "No recipes found. Check your spreadsheet headers."
                );
            }

            document.getElementById("uploadStatus").textContent =
                "Successfully loaded " + recipes.length + " recipes!";

            buildTagFilters();
            renderRecipes();

        } catch (error) {
            document.getElementById("uploadStatus").textContent =
                "Error: " + error.message;
        }
    };

    reader.readAsArrayBuffer(file);
});

// Create tag filter buttons.
function buildTagFilters() {

    const allTags = [...new Set(
        recipes.flatMap(r => r.tags)
    )].sort((a,b) => a.localeCompare(b));

    const container = document.getElementById("tagFilters");
    container.innerHTML = "";

    allTags.forEach(tag => {

        const button = document.createElement("button");
        button.className = "tag-option";
        button.textContent = tag;

        button.onclick = () => {

            if (selectedTags.has(tag)) {
                selectedTags.delete(tag);
                button.classList.remove("active");
            } else {
                selectedTags.add(tag);
                button.classList.add("active");
            }

            renderRecipes();
        };

        container.appendChild(button);
    });

    if (!allTags.length) {
        container.textContent = "No tags found in spreadsheet.";
    }
}

// Get recipes matching search and filters.
function getFilteredRecipes() {

    const search = document.getElementById("searchInput")
        .value.toLowerCase().trim();

    return recipes.filter(recipe => {

        const searchable = [
            recipe.name,
            recipe.ingredients,
            recipe.description,
            recipe.instructions,
            recipe.tags.join(" ")
        ].join(" ").toLowerCase();

        const matchesSearch = searchable.includes(search);

        // All selected tags must match.
        const matchesTags = [...selectedTags].every(tag =>
            recipe.tags.includes(tag)
        );

        return matchesSearch && matchesTags;
    });
}

// Display recipe cards.
function renderRecipes() {

    const filtered = getFilteredRecipes();
    const grid = document.getElementById("recipeGrid");

    document.getElementById("recipeCount").textContent =
        filtered.length + " recipes found";

    document.getElementById("filterCount").textContent =
        selectedTags.size
        ? selectedTags.size + " active tag filters"
        : "Showing all matching recipes";

    grid.innerHTML = "";

    if (!filtered.length) {
        grid.innerHTML = `
            <div class="empty">
                <h2>No recipes found</h2>
                <p>Try changing your search or filters.</p>
            </div>
        `;
        return;
    }

    filtered.forEach(recipe => {

        const card = document.createElement("article");
        card.className = "recipe-card";

        const title = document.createElement("h3");
        title.textContent = recipe.name;

        const description = document.createElement("p");
        description.className = "recipe-description";
        description.textContent =
            recipe.description || "Click to view this recipe.";

        const meta = document.createElement("p");
        meta.className = "recipe-description";

        const details = [];
        if (recipe.prepTime) details.push(recipe.prepTime);
        if (recipe.servings) details.push(recipe.servings + " servings");
        meta.textContent = details.join(" • ");

        const tags = document.createElement("div");
        tags.className = "recipe-tags";

        recipe.tags.forEach(tag => {
            const span = document.createElement("span");
            span.className = "recipe-tag";
            span.textContent = tag;
            tags.appendChild(span);
        });

        card.append(title, description, meta, tags);
        card.onclick = () => showRecipe(recipe);

        grid.appendChild(card);
    });
}

// Open recipe details.
function showRecipe(recipe) {

    const details = document.getElementById("recipeDetails");
    details.innerHTML = "";

    function addSection(title, content) {
        const heading = document.createElement("h3");
        heading.textContent = title;

        const text = document.createElement("div");
        text.className = "recipe-content";
        text.textContent = content || "Not provided.";

        details.append(heading, text);
    }

    const title = document.createElement("h2");
    title.textContent = recipe.name;
    details.appendChild(title);

    if (recipe.tags.length) {
        const tags = document.createElement("p");
        tags.textContent = recipe.tags.join(" • ");
        details.appendChild(tags);
    }

    if (recipe.prepTime) {
        addSection("Time", recipe.prepTime);
    }

    if (recipe.servings) {
        addSection("Servings", recipe.servings);
    }

    addSection("Ingredients", recipe.ingredients);
    addSection("Instructions", recipe.instructions);

    if (recipe.description) {
        addSection("Notes", recipe.description);
    }

    document.getElementById("recipeModal").style.display = "block";
}

function closeModal() {
    document.getElementById("recipeModal").style.display = "none";
}

function closeOutside(event) {
    if (event.target.id === "recipeModal") {
        closeModal();
    }
}

// Pick a random recipe from currently filtered results.
function pickRandom() {

    const filtered = getFilteredRecipes();

    if (!filtered.length) {
        document.getElementById("randomStatus").textContent =
            "No recipes match your current filters!";
        return;
    }

    const randomIndex = Math.floor(
        Math.random() * filtered.length
    );

    const recipe = filtered[randomIndex];

    document.getElementById("randomStatus").textContent =
        "Tonight's choice: " + recipe.name;

    showRecipe(recipe);
}

// Reset all filters.
function clearFilters() {

    selectedTags.clear();

    document.getElementById("searchInput").value = "";

    buildTagFilters();
    renderRecipes();

    document.getElementById("randomStatus").textContent = "";
}
